cmake_minimum_required(VERSION 3.21)
project(build_comparison LANGUAGES C)
option(ENABLE_ASAN "Enable AddressSanitizer" OFF)
add_executable(cmake_app main.c)
target_compile_options(cmake_app PRIVATE -Wall -Wextra)
if(ENABLE_ASAN)
    target_compile_options(cmake_app PRIVATE -fsanitize=address)
    target_link_options(cmake_app PRIVATE -fsanitize=address)
endif()
